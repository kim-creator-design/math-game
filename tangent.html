현재 관측 고도: 준비 중...


목표: 레이저가 지구와 90도(접선)가 될 때 정지하세요!
if (sessionStorage.getItem("tangentUnlocked") === "true") { let canvas = document.getElementById("gameCanvas"); let ctx = canvas.getContext("2d"); let isPlaying = false; let currentAngle = 10; let angleSpeed = 1.3; let animId; let realHeight, canvasHeight, personY, dist_OP, targetAngle; const earthX = 300, earthY = 450, earthR = 200; // 동탄 기준 (남북한 주요 도시 및 명산 포함) function getVisibleCity(km) { if (km < 40) return "수원, 오산, 서울(강남), 용인"; if (km < 80) return "인천, 천안, 원주, [북한] 개성"; if (km < 120) return "대전, 세종, 청주, [북한] 해주"; if (km < 170) return "전주, 안동, [북한] 사리원, 금강산"; if (km < 210) return "대구, 포항, [북한] 평양, 원산, 남포"; if (km < 240) return "광주, 진주, [북한] 안주, 함흥 입구"; return "여수, 목포, [북한] 묘향산 부근"; } function initGame() { // Math.pow(Math.random(), 2)를 사용하여 낮은 고도 확률 증가 (100m~5000m) let weightedRandom = Math.pow(Math.random(), 2); realHeight = Math.floor(weightedRandom * 4901) + 100; document.getElementById("heightDisplay").innerHTML = `현재 관측 고도: ${realHeight} m`; document.getElementById("msg").innerHTML = "목표: 레이저가 지구와 딱 스치는 접선(90도)이 될 때 정지하세요!"; document.getElementById("msg").style.color = "#4cd137"; document.getElementById("actionBtn").innerText = "정지! 🛑"; document.getElementById("actionBtn").style.background = "#e74c3c"; canvasHeight = 20 + (realHeight / 5000) * 80; personY = earthY - earthR - canvasHeight; dist_OP = earthY - personY; targetAngle = Math.asin(earthR / dist_OP) * (180 / Math.PI); currentAngle = 10; isPlaying = true; cancelAnimationFrame(animId); draw(); } function draw() { ctx.clearRect(0, 0, 600, 400); // 지구 ctx.beginPath(); ctx.arc(earthX, earthY, earthR, 0, Math.PI * 2); ctx.fillStyle = "#3498db"; ctx.fill(); // 관측 타워 ctx.fillStyle = "#95a5a6"; ctx.fillRect(earthX - 5, personY, 10, canvasHeight); ctx.fillStyle = "#ff4757"; ctx.beginPath(); ctx.arc(earthX, personY, 6, 0, Math.PI*2); ctx.fill(); // 레이저 let rad = currentAngle * (Math.PI / 180); let endX = earthX + 700 * Math.sin(rad); let endY = personY + 700 * Math.cos(rad); ctx.beginPath(); ctx.moveTo(earthX, personY); ctx.lineTo(endX, endY); ctx.strokeStyle = "#ff4757"; ctx.lineWidth = 3; if(!isPlaying) ctx.setLineDash([10, 5]); ctx.stroke(); ctx.setLineDash([]); if (isPlaying) { currentAngle += angleSpeed; if (currentAngle > 85 || currentAngle < 5) angleSpeed *= -1; animId = requestAnimationFrame(draw); } else { if (Math.abs(currentAngle - targetAngle) <= 2) { let R = 6371; let h = realHeight / 1000; let dist = Math.sqrt(Math.pow(R + h, 2) - Math.pow(R, 2)); let city = getVisibleCity(dist); document.getElementById("msg").innerHTML = `🎉 정확한 접선입니다!
` + `가시거리는 약 ${dist.toFixed(1)} km 이며,
` + `동탄에서 [${city}] 까지 보입니다!`; } else { document.getElementById("msg").innerHTML = `❌ 실패! 레이저가 지구를 뚫고 가거나 허공을 향했습니다.
(팁: 지구 표면을 정확히 스쳐야 합니다)`; } } } window.toggleGame = function() { if (isPlaying) { isPlaying = false; document.getElementById("actionBtn").innerText = "새로운 고도에서 다시 하기 🔄"; document.getElementById("actionBtn").style.background = "#2ecc71"; } else { initGame(); } }; initGame(); }
